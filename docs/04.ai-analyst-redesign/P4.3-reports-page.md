async function saveTemplate(formData) {
    const method = formData.id ? 'PUT' : 'POST';
    const url = formData.id
        ? `${API_BASE}/templates/${formData.id}`
        : `${API_BASE}/templates`;
    await fetch(url, {
        method,
        headers: { 'Content-Type': 'application/json', 'Authorization': getAuthHeader() },

// 立即执行报告
async function runReport(templateId) {
    const res = await fetch(`${API_BASE}/run/${templateId}`, {
        method: 'POST',
        headers: { 'Authorization': getAuthHeader() }
    });
    const data = await res.json();
    if (data.ok) alert('报告执行成功');
    else alert('执行失败: ' + data.error);
}

    openTab('report-view', `template_id=${templateId}`);

## 注意事项

1. **关联规则下拉菜单：** 使用 `GET /api/extract/rules?enabled=1` 获取已启用规则列表
2. **Prompt 模板默认值：** 新建时默认填入 `DEFAULT_REPORT_PROMPT` 的内容（后端 API 已有默认值）
3. **聚合维度与指标字段联动：** 选择"按厂商"后，指标字段的下拉菜单只显示该规则中的非 company 类型字段
