### Step 3：前端 JS 实现要点

```javascript
document.addEventListener('DOMContentLoaded', () => {
    loadTemplates();
});

// 加载可执行的报告模板列表
async function loadTemplates() {
    const res = await fetch('/api/reports/templates?enabled=1');
    const data = await res.json();
    const select = document.getElementById('template-select');
    data.forEach(t => {
        const opt = document.createElement('option');
        opt.value = t.id;
        opt.text = `${t.name} (${t.domain})`;
        select.appendChild(opt);
    });
}

// 立即执行
async function runNow() {
    const templateId = document.getElementById('template-select').value;
    const lookback = document.querySelector('input[name="lookback"]:checked').value;

    // 显示进度区域
    showProgress();

    try {
        const res = await fetch(`/api/reports/run/${templateId}`, {
            method: 'POST',
            headers: { 'Authorization': getAuthHeader() },
            body: JSON.stringify({ lookback_days: parseInt(lookback) })
        });
        const data = await res.json();

        if (data.ok) {
            hideProgress();
            showResult(data.run_id);
        } else {
            hideProgress();
            showError(data.error);
        }
    } catch (e) {
        hideProgress();
        showError('网络错误: ' + e.message);
    }
}

// 显示进度
function showProgress() {
    document.getElementById('progress-section').style.display = 'block';
    document.getElementById('result-section').style.display = 'none';
    // 模拟进度更新（轮询 run 状态）
    startProgressPoll();
}

// 轮询执行状态
async function startProgressPoll() {
    const templateId = document.getElementById('template-select').value;
    const pollInterval = setInterval(async () => {
        const res = await fetch(`/api/reports/runs/${templateId}?limit=1`);
        const data = await res.json();
        if (data.runs && data.runs[0]) {
            const latest = data.runs[0];
            if (latest.status === 'completed') {
                clearInterval(pollInterval);
                hideProgress();
                showResult(latest.id);
            } else if (latest.status === 'failed') {
                clearInterval(pollInterval);
                hideProgress();
                showError(latest.error_msg);
            }
        }
    }, 3000);
}

// 显示结果
async function showResult(runId) {
    document.getElementById('result-section').style.display = 'block';
    const res = await fetch(`/api/reports/runs/${runId}`);
    const run = await res.json();
    // 渲染结果（复用 report-view 的渲染逻辑）
    renderResult(run);
}

function renderResult(run) {
    // 渲染 Markdown
    document.getElementById('result-analysis').innerHTML = marked.parse(run.output_analysis || '');
    // 渲染图表
    if (run.output_charts && run.output_charts.length > 0) {
        renderCharts(run.output_charts, 'result-charts');
    }
    // 渲染总结
    document.getElementById('result-summary').textContent = run.output_summary || '';
}
```
