```python
"""Report scheduler: APScheduler-based background task runner."""
import json
import logging
import signal
import sys
import time
from datetime import datetime, timezone
from apscheduler.schedulers.background import BackgroundScheduler
from apscheduler.triggers.interval import IntervalTrigger

from core.db import get_db

logger = logging.getLogger(__name__)

scheduler = None  # 全局调度器实例


def start_scheduler():
    """启动调度器（作为独立进程的主入口）"""
    global scheduler

    scheduler = BackgroundScheduler(daemon=True)
    scheduler.add_job(
        func=_run_extract_cycle,
        trigger=IntervalTrigger(minutes=10),
        id="extract_cycle",
        replace_existing=True,
        misfire_grace_time=60,
    )
    scheduler.add_job(
        func=_run_report_cycle,
        trigger=IntervalTrigger(minutes=5),
        id="report_cycle",
        replace_existing=True,
        misfire_grace_time=60,
    )

    # 启动时执行一次（可选，用于首次加载）
    logger.info("Scheduler starting...")
    scheduler.start()
    logger.info("Scheduler started (extract=10min, report=5min)")

    # 注册信号处理，优雅退出
    def _signal_handler(signum, frame):
        logger.info("Scheduler shutting down...")
        scheduler.shutdown(wait=False)
        sys.exit(0)

    signal.signal(signal.SIGTERM, _signal_handler)
    signal.signal(signal.SIGINT, _signal_handler)

    # 保持主线程存活
    try:
        while True:
            time.sleep(60)
    except (KeyboardInterrupt, SystemExit):
        logger.info("Scheduler stopped")


def _run_extract_cycle():
    """抽取任务轮询（每 10 分钟执行一次）"""
    from core.db import get_setting, get_db_path
    from core.scheduler.extraction_engine import extract_all_pending

    # 检查开关
    enabled = get_setting("scheduler.extract_enabled", "1")
    if enabled == "0":
        return

    interval_min = int(get_setting("scheduler.extract_interval_min", "10"))
    logger.info(f"[extract_cycle] Running extraction cycle (interval={interval_min}min)")

    try:
        result = extract_all_pending()
        logger.info(f"[extract_cycle] Done: {result}")

        # 更新状态
        _update_scheduler("last_extract_time", datetime.now(timezone.utc).isoformat())

    except Exception as e:
        logger.error(f"[extract_cycle] Error: {e}")


def _run_report_cycle():
    """报告任务轮询（每 5 分钟执行一次）"""
    from core.db import get_setting, get_db_path
    from core.scheduler.report_engine import run_scheduled_reports

    # 检查开关
    enabled = get_setting("scheduler.report_enabled", "1")
    if enabled == "0":
        return

    interval_min = int(get_setting("scheduler.report_interval_min", "5"))
    logger.info(f"[report_cycle] Running report cycle (interval={interval_min}min)")

    try:
        result = run_scheduled_reports()
        logger.info(f"[report_cycle] Done: {result}")

        _update_scheduler("last_report_time", datetime.now(timezone.utc).isoformat())

    except Exception as e:
        logger.error(f"[report_cycle] Error: {e}")


def _update_scheduler(field: str, value):
    """更新 report_scheduler 状态"""
    from core.db import get_db_path

    db_path = get_db_path()
    now = datetime.now(timezone.utc).isoformat()
    with get_db(db_path) as conn:
        conn.execute(
            f"UPDATE report_scheduler SET {field} = ?, updated_at = ? WHERE id = 1",
            (value, now)
        )
        conn.commit()


# ─── 手动触发接口 ─────────────────────────────────────────

def trigger_extract_once():
    """手动触发一次抽取（供 API 调用）"""
    from core.scheduler.extraction_engine import extract_all_pending
    return extract_all_pending()


def trigger_report_once(template_id: int):
    """手动触发一个报告的执行（供 API 调用）"""
    from core.scheduler.report_engine import run_single_report
    return run_single_report(template_id)


def trigger_report_all():
    """手动触发所有到期报告的执行（供 API 调用）"""
    from core.scheduler.report_engine import run_scheduled_reports
    return run_scheduled_reports()
```

## 注意事项

1. **daemon=True：** APScheduler 的 BackgroundScheduler 设为 daemon 模式，确保主线程退出时自动清理
2. **misfire_grace_time=60：** 任务错过触发时间 60 秒内仍会补触发，防止重启后任务堆积
3. **主线程 sleep(60)：** 保持进程存活，实际生产环境建议使用更健壮的方案（如 `threading.Event`）
4. **`run_extract_cycle` 和 `run_report_cycle` 中延迟导入：** 避免循环导入问题
