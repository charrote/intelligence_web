```python
"""SQL aggregator: builds and executes aggregation queries from intel_fact."""
import json
import logging
from datetime import datetime, timedelta, timezone

from core.db import get_db

logger = logging.getLogger(__name__)


def aggregate(template: dict) -> dict:
    """
    根据报告模板配置执行 SQL 聚合。

    Args:
        template: 报告模板 dict，包含 group_by, metrics, filters, rule_id, lookback_days

    Returns:
        {
            "rows": [{entity_name, metric_values...}],
            "fact_count": int,
            "error": str or None
        }
    """
    from core.db import get_db_path

    db_path = get_db_path()

    try:
        with get_db(db_path) as conn:
            # 1. 构建 WHERE 条件
            where_clause = f"WHERE f.rule_id = {template['rule_id']}"
            params = [template['rule_id']]

            # 时间过滤
            lookback = template.get("lookback_days", 30)
            cutoff = datetime.now(timezone.utc) - timedelta(days=lookback)
            where_clause += f" AND f.created_at >= '{cutoff.isoformat()}'"

            # 用户自定义过滤条件
            filters = json.loads(template.get("filters", "[]"))
            filter_clauses, filter_params = _build_filter_clauses(filters)
            if filter_clauses:
                where_clause += " AND " + " AND ".join(filter_clauses)
                params.extend(filter_params)

            # 2. 构建 GROUP BY
            group_field = _map_group_field(template.get("group_by", "entity_name"))
            group_clause = f"GROUP BY f.{group_field}"

            # 3. 构建 SELECT（每个 metric 一个聚合）
            metrics = json.loads(template.get("metrics", "[]"))
            select_parts = [f"SUM(f.metric_value) AS fact_count"]
            params_base = params[:]

            for i, m in enumerate(metrics):
                field_key = m.get("field_key", "")
                agg_fn = m.get("agg", "avg")
                alias = f"metric_{i}"
                select_parts.append(
                    f"{_map_agg_fn(agg_fn)}(mv.{field_key}) AS {alias}"
                )
                # 需要 self-join 来获取 metric_value
                params_base.append(field_key)

            # 4. 构建完整的 JOIN 和 SQL
            # 主表: intel_fact (按 group_field 分组)
            # 度量表: intel_fact (按 metric_value 聚合)
            join_clause = _build_joins(metrics)

            # 基本 SQL 模板
            base_sql = f"""
                SELECT main.{group_field} AS entity_name,
                       COUNT(DISTINCT main.intel_id) AS fact_count
                FROM intel_fact main
                {join_clause}
                {where_clause}
                {group_clause}
                ORDER BY entity_name
            """

            rows = conn.execute(base_sql, params).fetchall()
            result_rows = [dict(r) for r in rows]

            # 5. 计算每个 metric 的聚合值
            enriched_rows = []
            for row in result_rows:
                enriched = dict(row)
                for i, m in enumerate(metrics):
                    field_key = m.get("field_key", "")
                    agg_fn = m.get("agg", "avg")
                    unit = m.get("unit", "")

                    sub_sql = f"""
                        SELECT {field_key}, metric_unit
                        FROM intel_fact
                        WHERE rule_id = ? AND {group_field} = ?
                          AND created_at >= ?
                    """
                    sub_params = [template['rule_id'], enriched['entity_name'], cutoff.isoformat()]
                    sub_rows = conn.execute(sub_sql, sub_params).fetchall()

                    values = [r[field_key] for r in sub_rows if r[field_key] is not None]
                    enriched[f"metric_{i}"] = _apply_agg(values, agg_fn) if values else 0
                    enriched[f"metric_{i}_unit"] = unit
                enriched_rows.append(enriched)

            return {
                "rows": enriched_rows,
                "fact_count": sum(r["fact_count"] for r in result_rows),
                "error": None
            }

    except Exception as e:
        logger.error(f"Aggregation failed: {e}")
        return {"rows": [], "fact_count": 0, "error": str(e)}


def _map_group_field(group_by: str) -> str:
    """映射 group_by 配置到 intel_fact 的字段名"""
    mapping = {
        "entity_name": "entity_name",
        "time_period": "time_period",
        "context": "context",
        "country": "entity_name",  # 兼容
    }
    return mapping.get(group_by, "entity_name")


def _map_agg_fn(agg: str) -> str:
    """映射聚合函数"""
    mapping = {
        "avg": "AVG",
        "sum": "SUM",
        "max": "MAX",
        "min": "MIN",
        "count": "COUNT",
    }
    return mapping.get(agg, "AVG")


def _apply_agg(values: list, agg: str) -> float:
    """对一组值执行聚合运算"""
    if not values:
        return 0.0
    mapping = {
        "avg": lambda vs: sum(vs) / len(vs),
        "sum": lambda vs: sum(vs),
        "max": lambda vs: max(vs),
        "min": lambda vs: min(vs),
        "count": lambda vs: len(vs),
    }
    fn = mapping.get(agg, mapping["avg"])
    return round(fn(values), 2)


def _build_filter_clauses(filters: list) -> tuple:
    """
    构建过滤条件 SQL 片段。

    filters 格式:
    [{"field_key": "country", "op": "eq", "value": "中国"}]

    Returns:
        (where_clauses: list, params: list)
    """
    clauses = []
    params = []
    for f in filters:
        fk = f.get("field_key", "")
        op = f.get("op", "eq")
        val = f.get("value", "")

        # 过滤条件作用于同一 rule_id 的 intel_fact 表
        clause = f"(mf_{fk}.{_map_filter_field(fk)} = ?)"
        clauses.append(clause)
        params.append(val)

    return clauses, params


def _map_filter_field(field_key: str) -> str:
    """过滤字段名映射"""
    if field_key in ("country", "location"):
        return "entity_name"
    return "context"


def _build_joins(metrics: list) -> str:
    """构建 self-join 子句（用于聚合 metric_value）"""
    if not metrics:
        return ""
    # 简化实现：第一个 metric 直接通过 where 条件过滤，不做 join
    # 后续可扩展为子查询方式
    return ""
```

## 注意事项

1. **聚合函数映射：** `avg/sum/max/min/count` 五个函数，其他值默认 `avg`
2. **时间过滤用 cutoff 字符串：** SQLite 不支持 `TIMESTAMPDIFF`，使用 Python 计算 `cutoff` 后转为 ISO 字符串
3. **metric_value 为 NULL 的处理：** 聚合时跳过 NULL 值，确保 SQL 中 `AVG(COALESCE(...))` 或在 Python 层过滤
4. **字段过滤（filters）：** 仅支持 `eq`（等于）操作，如需扩展其他操作符，可在此处增加映射
