### Step 3：前端 JS 实现要点

```javascript
// 从 URL 参数获取 template_id
const params = new URLSearchParams(window.location.search);
const templateId = params.get('template_id') || 1;

document.addEventListener('DOMContentLoaded', () => {
    loadRunHistory();
});

// 加载执行历史
async function loadRunHistory() {
    const res = await fetch(`/api/reports/runs/${templateId}`);
    const data = await res.json();
    renderHistoryDropdown(data.runs);

    // 自动加载最新一次执行
    if (data.runs.length > 0) {
        loadRunDetail(data.runs[0].id);
    }
}

// 渲染版本切换下拉
function renderHistoryDropdown(runs) {
    const select = document.getElementById('run-select');
    runs.forEach(run => {
        const opt = document.createElement('option');
        opt.value = run.id;
        const statusIcon = run.status === 'completed' ? '✅' : '❌';
        opt.text = `${run.scheduled_time} ${statusIcon}`;
        select.appendChild(opt);
    });
    select.addEventListener('change', (e) => loadRunDetail(e.target.value));
}

// 加载单次报告详情
async function loadRunDetail(runId) {
    const res = await fetch(`/api/reports/runs/${runId}`);
    const run = await res.json();

    if (run.error) {
        document.getElementById('error-section').style.display = 'block';
        document.getElementById('error-msg').textContent = run.error_msg;
        return;
    }

    // 渲染统计摘要
    document.getElementById('fact-count').textContent = run.fact_count;
    document.getElementById('duration').textContent = `${run.duration_sec}s`;

    // 渲染 Markdown 分析文字（使用 marked.js）
    const analysis = run.output_analysis || '暂无分析文字';
    document.getElementById('analysis-content').innerHTML = marked.parse(analysis);

    // 渲染图表
    renderCharts(run.output_charts);

    // 渲染一段话总结
    document.getElementById('summary-text').textContent = run.output_summary || '';
}

// 渲染 ECharts 图表
function renderCharts(charts) {
    const container = document.getElementById('charts-container');
    container.innerHTML = '';

    charts.forEach((chart, index) => {
        const div = document.createElement('div');
        div.className = 'chart-box';
        div.innerHTML = `<h4>${chart.title}</h4><div id="chart-${index}" style="height:300px;"></div>`;
        container.appendChild(div);

        if (chart.type === 'table') {
            renderTable(chart, document.getElementById(`chart-${index}`));
        } else {
            const option = buildEChartsOption(chart);
            const chartInstance = echarts.init(document.getElementById(`chart-${index}`));
            chartInstance.setOption(option);
        }
    });
}

function buildEChartsOption(chart) {
    const data = chart.data || [];
    const names = data.map(d => d.name);
    const values = data.map(d => d.value);

    if (chart.type === 'pie') {
        return {
            title: { text: chart.title, left: 'center' },
            tooltip: { trigger: 'item' },
            series: [{ type: 'pie', data: data }]
        };
    }
    if (chart.type === 'bar') {
        return {
            title: { text: chart.title },
            xAxis: { type: 'category', data: names },
            yAxis: { type: 'value' },
            series: [{ type: 'bar', data: values }]
        };
    }
    if (chart.type === 'line') {
        return {
            title: { text: chart.title },
            xAxis: { type: 'category', data: names },
            yAxis: { type: 'value' },
            series: [{ type: 'line', data: values, smooth: true }]
        };
    }
    return {};
}

function renderTable(chart, container) {
    const data = chart.data || [];
    let html = '<table><thead><tr><th>名称</th><th>数值</th></tr></thead><tbody>';
    data.forEach(d => {
        html += `<tr><td>${d.name}</td><td>${d.value}</td></tr>`;
    });
    html += '</tbody></table>';
    container.innerHTML = html;
}
```

## 注意事项

1. **ECharts 图表自适应：** 使用 `window.addEventListener('resize')` 重新调整图表尺寸
2. **表格渲染在 chart-box 内：** 与图表同一容器，表格用 CSS table 样式
3. **Markdown 安全：** `marked.parse()` 可能执行 HTML，如安全风险可考虑 sanitize
