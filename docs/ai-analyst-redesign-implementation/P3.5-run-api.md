### Step 1：在 `core/app.py` 新增路由

```python
# ─── 报告执行与查看 API ────────────────────────────────────

@app.route('/api/reports/run/<int:template_id>', methods=['POST'])
@require_auth
def run_report(template_id):
    """手动触发单次报告执行"""
    result = trigger_report_once(template_id)
    return jsonify(result)


@app.route('/api/reports/runs/<int:template_id>', methods=['GET'])
@require_auth
def list_report_runs(template_id):
    """获取某模板的执行历史（分页）"""
    from core.db import get_db_path
    db_path = get_db_path()
    limit = min(int(request.args.get('limit', 20)), 100)
    offset = int(request.args.get('offset', 0))

    with get_db(db_path) as conn:
        total = conn.execute(
            "SELECT COUNT(*) FROM report_run WHERE template_id = ?",
            (template_id,)
        ).fetchone()[0]

        rows = conn.execute(
            """SELECT id, template_id, domain, scheduled_time, completed_at,
                      status, duration_sec, retry_count, fact_count
               FROM report_run
               WHERE template_id = ?
               ORDER BY scheduled_time DESC
               LIMIT ? OFFSET ?""",
            (template_id, limit, offset)
        ).fetchall()

        return jsonify({
            "total": total,
            "limit": limit,
            "offset": offset,
            "runs": [dict(r) for r in rows]
        })


@app.route('/api/reports/runs/<int:run_id>', methods=['GET'])
@require_auth
def get_report_run(run_id):
    """获取单次报告详情"""
    from core.db import get_db_path
    db_path = get_db_path()
    with get_db(db_path) as conn:
        run = conn.execute(
            """SELECT r.*, a.name AS template_name, a.domain AS template_domain
               FROM report_run r
               LEFT JOIN intel_aggregate a ON r.template_id = a.id
               WHERE r.id = ?""",
            (run_id,)
        ).fetchone()

        if not run:
            return jsonify({"error": "Not found"}), 404

        d = dict(run)
        # 解析 JSON 字段
        for key in ("aggregated_data", "output_charts"):
            if d.get(key):
                d[key] = json.loads(d[key])

        return jsonify(d)


@app.route('/api/reports/overview', methods=['GET'])
@require_auth
def report_overview():
    """获取所有模板的最新一次执行结果摘要"""
    from core.db import get_db_path
    db_path = get_db_path()
    with get_db(db_path) as conn:
        # 获取每个模板的最新一次执行
        rows = conn.execute(
            """SELECT a.id AS template_id, a.name, a.domain,
                      r.status AS last_status,
                      r.completed_at AS last_time,
                      r.duration_sec AS last_duration,
                      r.fact_count AS last_fact_count,
                      a.status AS template_status
               FROM intel_aggregate a
               LEFT JOIN report_run r ON a.id = r.template_id
               AND r.id = (
                   SELECT MAX(id) FROM report_run WHERE template_id = a.id
               )
               WHERE a.enabled = 1
               ORDER BY a.domain, a.name"""
        ).fetchall()

        return jsonify([dict(r) for r in rows])
```

## 验收标准

| # | 验收项 | 验证方式 |
|---|--------|----------|
| 1 | `POST /api/reports/run/{template_id}` 触发报告 | 验证 `report_run` 有新记录 |
| 2 | `GET /api/reports/runs/{template_id}` 返回执行历史 | 验证分页 + 排序正确 |
| 3 | `GET /api/reports/runs/{run_id}` 返回详情 | 验证 JSON 字段已解析 |
| 4 | `GET /api/reports/overview` 返回所有模板最新状态 | 验证每个模板一条记录 |
| 5 | 查询不存在的 run_id 返回 404 | 验证错误处理 |

---

## 注意事项

1. **`output_analysis` 字段：** 在 `get_report_run` 中直接返回 TEXT 字段，无需解析 JSON
2. **`aggregated_data` 和 `output_charts` 字段：** 是 JSON 字符串，读取时用 `json.loads()` 解析
3. **`overview` 的性能：** 使用子查询 `MAX(id)` 获取每个模板的最新记录，数据量大时可能慢，可考虑加索引 `idx_run_template ON report_run(template_id)`
