### Step 3：前端 JS 实现

```javascript
// 加载调度器配置
async function loadSchedulerConfig() {
    const res = await fetch('/api/reports/scheduler');
    const data = await res.json();

    document.getElementById('extract-enabled').checked = data.extract_enabled === 1;
    document.getElementById('report-enabled').checked = data.report_enabled === 1;
    document.getElementById('extract-interval').value = data.extract_interval_min;
    document.getElementById('report-interval').value = data.report_interval_min;
    document.getElementById('extract-timeout').value = data.extract_timeout || 60;
    document.getElementById('report-timeout').value = data.report_timeout || 120;
    document.getElementById('retry-count').value = data.retry_count || 1;
    document.getElementById('fuse-threshold').value = data.fuse_threshold || 3;
    document.getElementById('retention-days').value = data.retention_days || 30;

    // 显示状态
    const status = document.getElementById('scheduler-status');
    const extractIcon = data.extract_enabled ? '🟢' : '🔴';
    const reportIcon = data.report_enabled ? '🟢' : '🔴';
    status.innerHTML = `<span>${extractIcon} 抽取: ${data.extract_enabled ? '运行' : '暂停'}</span>
                        <span>${reportIcon} 报告: ${data.report_enabled ? '运行' : '暂停'}</span>
                        <span>上次抽取: ${data.last_extract_time || '从未'}</span>
                        <span>上次报告: ${data.last_report_time || '从未'}</span>`;
}

// 保存调度器配置
async function saveSchedulerConfig() {
    const config = {
        extract_enabled: document.getElementById('extract-enabled').checked ? 1 : 0,
        report_enabled: document.getElementById('report-enabled').checked ? 1 : 0,
        extract_interval_min: document.getElementById('extract-interval').value,
        report_interval_min: document.getElementById('report-interval').value,
        extract_timeout: document.getElementById('extract-timeout').value,
        report_timeout: document.getElementById('report-timeout').value,
        retry_count: document.getElementById('retry-count').value,
        fuse_threshold: document.getElementById('fuse-threshold').value,
        retention_days: document.getElementById('retention-days').value,
    };

    // 保存 LLM 超时设置到 settings 表
    await fetch('/api/settings', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Authorization': getAuthHeader() },
        body: JSON.stringify([
            { key: 'llm.extract_timeout', value: config.extract_timeout },
            { key: 'llm.report_timeout', value: config.report_timeout },
            { key: 'report.max_retries', value: config.retry_count },
            { key: 'report.fuse_threshold', value: config.fuse_threshold },
            { key: 'report.retention_days', value: config.retention_days },
        ])
    });

    // 开关调度器
    await fetch('/api/reports/scheduler/toggle', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Authorization': getAuthHeader() },
        body: JSON.stringify({
            action: config.extract_enabled ? 'enable' : 'disable',
            component: 'extract'
        })
    });

    await fetch('/api/reports/scheduler/toggle', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Authorization': getAuthHeader() },
        body: JSON.stringify({
            action: config.report_enabled ? 'enable' : 'disable',
            component: 'report'
        })
    });

    alert('配置已保存');
    loadSchedulerConfig();
}

async function runAllReports() {
    await fetch('/api/reports/scheduler/run-all', {
        method: 'POST',
        headers: { 'Authorization': getAuthHeader() }
    });
    alert('报告执行任务已提交');
    loadSchedulerConfig();
}

// 页面加载时加载调度器配置
document.addEventListener('DOMContentLoaded', () => {
    loadSchedulerConfig();
});
```
